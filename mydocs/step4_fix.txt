import numpy as np
from models.schemas import PatentRecord, ExtractedTokens

def compute_token_coverage(
    patent: PatentRecord,
    tokens: ExtractedTokens,
) -> dict:
    """
    Compute how many critical token GROUPS are represented in the patent.
    
    The key insight: tokens from the SAME domain concept count as one group.
    "infrared" and "IR sensor" both satisfy the sensor concept — not two wins.
    
    Returns coverage metrics for scoring and diagnostics.
    """
    text = (
        f"{patent.patent_title or ''} {patent.patent_abstract or ''}"
    ).lower()

    # Check each domain concept independently
    # e.g. for "infrared lane detection autonomous vehicle":
    #   concept 1: sensor        → ["infrared", "IR", "thermal"]
    #   concept 2: detection     → ["lane detection", "road boundary"]
    #   concept 3: autonomous_vehicle → ["autonomous vehicle", "self-driving"]
    
    concept_hits = {}
    for concept, data in tokens.concept_groups.items():
        terms = [t.lower() for t in data.get("terms", [])]
        synonyms = [s.lower() for s in data.get("patent_synonyms", [])]
        all_terms = terms + synonyms
        hit = any(t in text for t in all_terms)
        concept_hits[concept] = hit

    total_concepts = len(concept_hits)
    matched_concepts = sum(concept_hits.values())
    coverage = matched_concepts / total_concepts if total_concepts else 0.0

    return {
        "coverage":         coverage,           # 0.0 – 1.0
        "matched_concepts": matched_concepts,
        "total_concepts":   total_concepts,
        "concept_hits":     concept_hits,       # for diagnostics
    }


def apply_coverage_penalty(
    hybrid_score: float,
    coverage: float,
    penalty_curve: str = "quadratic",
) -> float:
    """
    Apply a smooth penalty based on concept coverage.
    
    Coverage 1.0 (all concepts matched) → no penalty
    Coverage 0.67 (2/3 matched)         → moderate penalty  
    Coverage 0.33 (1/3 matched)         → heavy penalty
    Coverage 0.0  (nothing matched)     → maximum penalty
    
    penalty_curve options:
      linear    : penalty scales linearly with missing coverage
      quadratic : penalty accelerates — missing 2 concepts hurts much
                  more than 2x missing 1 concept (recommended)
      step      : hard cutoffs at thresholds (least nuanced)
    """
    if penalty_curve == "linear":
        multiplier = coverage                           # 0.33 → score * 0.33

    elif penalty_curve == "quadratic":
        multiplier = coverage ** 2                     # 0.33 → score * 0.11
                                                       # 0.67 → score * 0.45
                                                       # 1.00 → score * 1.00

    elif penalty_curve == "step":
        if coverage == 1.0:
            multiplier = 1.0
        elif coverage >= 0.67:
            multiplier = 0.6
        elif coverage >= 0.33:
            multiplier = 0.25
        else:
            multiplier = 0.05

    else:
        multiplier = coverage  # fallback to linear

    return hybrid_score * multiplier

import numpy as np
from retrieval.similarity import compute_token_coverage, apply_coverage_penalty
from models.schemas import PatentRecord, RankedPatent, ExtractedTokens
from utils.logger import get_logger

log = get_logger(__name__)

def hybrid_rank(
    patents:       list[PatentRecord],
    cosine_scores: np.ndarray,
    bm25_scores:   np.ndarray,
    tokens:        ExtractedTokens,           # ← now required
    cosine_weight: float = 0.6,
    bm25_weight:   float = 0.4,
) -> list[RankedPatent]:
    """
    Compute hybrid score then apply concept coverage penalty per patent.
    Patents matching only one concept (e.g. only 'infrared', not 
    'lane detection' or 'autonomous vehicle') are penalized heavily.
    """
    cos_norm = _normalize(cosine_scores)
    bm25_norm = _normalize(bm25_scores)
    raw_hybrid = cosine_weight * cos_norm + bm25_weight * bm25_norm

    ranked = []
    for i, patent in enumerate(patents):

        # Compute concept coverage for this patent
        coverage_data = compute_token_coverage(patent, tokens)
        coverage      = coverage_data["coverage"]

        # Apply smooth penalty
        penalized_score = apply_coverage_penalty(
            raw_hybrid[i],
            coverage,
            penalty_curve="quadratic",   # tune: try "step" if too aggressive
        )

        log.debug(
            "Patent scored",
            extra={
                "patent_id":        patent.patent_id,
                "title":            patent.patent_title[:60],
                "raw_hybrid":       round(float(raw_hybrid[i]), 4),
                "coverage":         round(coverage, 2),
                "concept_hits":     coverage_data["concept_hits"],
                "penalized_score":  round(penalized_score, 4),
            }
        )

        ranked.append(RankedPatent(
            patent          = patent,
            cosine_score    = float(cosine_scores[i]),
            bm25_score      = float(bm25_scores[i]),
            hybrid_score    = penalized_score,          # penalized, not raw
            coverage        = coverage,                 # expose in UI
            concept_hits    = coverage_data["concept_hits"],
        ))

    ranked.sort(key=lambda r: r.hybrid_score, reverse=True)
    return ranked


def _normalize(scores: np.ndarray) -> np.ndarray:
    mn, mx = scores.min(), scores.max()
    return (scores - mn) / (mx - mn + 1e-9)
