The Fix — use _text_phrase instead of _text_any
_text_phrase requires the tokens to appear together, giving you phrase matching rather than token-level OR. Here's the corrected _build_term_query:
pythondef _build_term_query(term: str) -> dict:
    """
    Use _text_phrase for precise matching instead of _text_any.
    _text_any matches patents containing ANY single token (too broad).
    _text_phrase requires tokens to appear as a phrase (domain-specific).
    """
    cleaned = _strip_stop_words(term)
    
    # Split into meaningful sub-phrases if the cleaned term is long
    tokens = cleaned.split()
    
    if len(tokens) <= 3:
        # Short query: single phrase match across title + abstract
        return {
            "_or": [
                {"_text_phrase": {"patent_title": cleaned}},
                {"_text_phrase": {"patent_abstract": cleaned}},
            ]
        }
    else:
        # Longer query: build AND of key bigrams to force co-occurrence
        # e.g. "lane detection infrared sensors autonomous vehicle"
        # → title/abstract must contain "lane detection" AND "infrared sensors"
        bigrams = [f"{tokens[i]} {tokens[i+1]}" for i in range(0, len(tokens)-1, 2)]
        and_clauses = []
        for bigram in bigrams[:3]:  # cap at 3 bigrams to avoid over-constraining
            and_clauses.append({
                "_or": [
                    {"_text_phrase": {"patent_title": bigram}},
                    {"_text_phrase": {"patent_abstract": bigram}},
                ]
            })
        return {"_and": and_clauses}

Fix the 2-char filter too — "IR" is a critical domain abbreviation that your current len(w) > 2 filter silently drops:
pythondef _strip_stop_words(term: str) -> str:
    tokens = [
        w for w in term.lower().split()
        if w not in _STOP_WORDS and len(w) > 1  # allow 2-char terms like IR, AI, UV
    ]
    cleaned = " ".join(tokens)
    if not cleaned:
        log.warning("Stop-word stripping left empty query; using original",
                    extra={"original": term})
        return term
    log.debug("Query cleaned", extra={"original": term, "cleaned": cleaned})
    return cleaned
