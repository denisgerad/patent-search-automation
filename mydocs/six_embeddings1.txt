Fix B — Vector store is being used as a corpus instead of a cache
This is the more critical fix. Your vector_store.py should never be the source of patents for ranking. It should only cache embeddings to avoid recomputing them. The ranking corpus must always come from the live search results:

class VectorStore:
    """
    Embedding CACHE only.
    
    CORRECT use:  store embeddings after computing them,
                  retrieve them to avoid recomputation on repeated queries.
    
    WRONG use:    querying the store as the patent search corpus.
                  The corpus always comes from search_service results.
    """
    def store_embeddings(
        self, 
        patents: list[PatentRecord], 
        embeddings: np.ndarray
    ) -> None:
        """Cache embeddings keyed by patent_id."""
        for patent, vec in zip(patents, embeddings):
            self._store[patent.patent_id] = vec

    def get_cached_embeddings(
        self, 
        patents: list[PatentRecord]
    ) -> tuple[list[PatentRecord], list[PatentRecord], np.ndarray]:
        """
        Split patents into cached and uncached.
        Returns: (cached_patents, uncached_patents, cached_vecs)
        """
        cached_patents, cached_vecs, uncached_patents = [], [], []
        for p in patents:
            if p.patent_id in self._store:
                cached_patents.append(p)
                cached_vecs.append(self._store[p.patent_id])
            else:
                uncached_patents.append(p)
        
        vecs = np.array(cached_vecs) if cached_vecs else np.empty((0, 1024))
        return cached_patents, uncached_patents, vecs
